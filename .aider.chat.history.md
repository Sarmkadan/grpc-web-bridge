using Microsoft.Extensions.Configuration;
using System.ComponentModel.DataAnnotations;

namespace GrpcWebBridge.Configuration
{
    public class GrpcWebBridgeOptions
    {
        /// <summary>
        /// Environment name (Development/Production)
        /// </summary>
        [Required]
        public string Environment { get; set; } = "Development";

        /// <summary>
        /// Service instance name for identification
        /// </summary>
        [Required]
        public string InstanceName { get; set; } = "grpc-web-bridge";

        /// <summary>
        /// Enable detailed logging
        /// </summary>
        public bool EnableLogging { get; set; } = true;

        /// <summary>
        /// Enable Swagger documentation
        /// </summary>
        public bool EnableSwagger { get; set; } = true;

        /// <summary>
        /// Enable metrics endpoint
        /// </summary>
        public bool EnableMetrics { get; set; } = true;

        /// <summary>
        /// Enable CORS support
        /// </summary>
        public bool EnableCors { get; set; } = true;

        /// <summary>
        /// Require authentication for all endpoints
        /// </summary>
        public bool RequireAuthentication { get; set; } = false;

        /// <summary>
        /// Maximum concurrent streams allowed
        /// </summary>
        [Range(1, 100000)]
        public int MaxStreamCount { get; set; } = 10000;

        /// <summary>
        /// Stream idle timeout in seconds
        /// </summary>
        [Range(1, 3600)]
        public int StreamIdleTimeoutSeconds { get; set; } = 300;

        /// <summary>
        /// Stream heartbeat interval in seconds
        /// </summary>
        [Range(1, 300)]
        public int StreamHeartbeatIntervalSeconds { get; set; } = 30;

        /// <summary>
        /// Maximum message size in bytes
        /// </summary>
        [Range(1, 10485760)]
        public int MaxMessageSize { get; set; } = 4194304;

        /// <summary>
        /// Default timeout for requests in milliseconds
        /// </summary>
        [Range(1, 300000)]
        public int DefaultTimeoutMilliseconds { get; set; } = 30000;

        /// <summary>
        /// Enable response compression
        /// </summary>
        public bool CompressResponses { get; set; } = true;

        /// <summary>
        /// Compression level (1-9)
        /// </summary>
        [Range(1, 9)]
        public int CompressionLevel { get; set; } = 6;

        /// <summary>
        /// List of allowed origins for CORS
        /// </summary>
        public string[] AllowedOrigins { get; set; } = new[] { "*" };

        /// <summary>
        /// List of allowed HTTP methods
        /// </summary>
        public string[] AllowedMethods { get; set; } = new[] { "GET", "POST", "PUT", "DELETE", "OPTIONS" };

        /// <summary>
        /// Configure and validate options from configuration
        /// </summary>
        public static GrpcWebBridgeOptions FromConfiguration(IConfiguration configuration)
        {
            var options = new GrpcWebBridgeOptions();
            configuration.GetSection("GrpcWebBridge").Bind(options);
            
            var context = new ValidationContext(options);
            var results = new List<ValidationResult>();
            
            if (!Validator.TryValidateObject(options, context, results, true))
            {
                var errors = string.Join("\n", results.Select(r => r.ErrorMessage));
                throw new InvalidOperationException($"Invalid GrpcWebBridge configuration:\n{errors}");
            }
            
            return options;
        }
    }
}
